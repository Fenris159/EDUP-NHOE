cmake_minimum_required(VERSION 3.20)
project(edup_nhoe_vr VERSION 0.12.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
get_filename_component(ED_ROOT "${CMAKE_CURRENT_LIST_DIR}/.." ABSOLUTE)
set(OPENVR_ROOT "${ED_ROOT}/third_party/openvr/0924064316de3effbcd1acf1e309182a2deb1c05")
set(OPENXR_ROOT "${ED_ROOT}/third_party/openxr/24a1f2b92d212501a2b1b8d30ffd53abf55fcfa8")
add_executable(ed_vr_haptics steamvr_bridge.cpp)
target_include_directories(ed_vr_haptics PRIVATE "${ED_ROOT}/include")
target_include_directories(ed_vr_haptics SYSTEM PRIVATE "${OPENVR_ROOT}/headers")
target_compile_options(ed_vr_haptics PRIVATE -Wall -Wextra -Wpedantic -Werror)
set_target_properties(ed_vr_haptics PROPERTIES OUTPUT_NAME ed-vr-haptics)
if(WIN32)
  target_link_libraries(ed_vr_haptics PRIVATE "${OPENVR_ROOT}/lib/win64/openvr_api.lib" ws2_32)
  target_compile_definitions(ed_vr_haptics PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_link_options(ed_vr_haptics PRIVATE -static -static-libgcc -static-libstdc++ -Wl,--no-insert-timestamp)
  configure_file("${OPENVR_ROOT}/bin/win64/openvr_api.dll" openvr_api.dll COPYONLY)
  # Build the Windows runtime with haptic-only VR mirroring.
  add_subdirectory("${ED_ROOT}" core EXCLUDE_FROM_ALL)
  add_library(ed_vr_openxr SHARED openxr_layer.cpp openxr_layer.def)
  trace_settings(ed_vr_openxr)
  target_include_directories(ed_vr_openxr PRIVATE "${ED_ROOT}/include")
  target_include_directories(ed_vr_openxr SYSTEM PRIVATE "${OPENXR_ROOT}/include")
  target_link_options(ed_vr_openxr PRIVATE -Wl,--exclude-all-symbols)
  set_target_properties(ed_vr_openxr PROPERTIES PREFIX "" OUTPUT_NAME ed-vr-openxr RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/openxr" ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/openxr")
  get_target_property(HOST_SOURCES dinput_restoration_proxy SOURCES)
  list(TRANSFORM HOST_SOURCES PREPEND "${ED_ROOT}/")
  list(REMOVE_ITEM HOST_SOURCES "${ED_ROOT}/src/dinput8.def")
  add_library(dinput_vr_proxy SHARED ${HOST_SOURCES} publisher_win.cpp bridge_exports.cpp dinput_vr.def)
  target_sources(dinput_vr_proxy PRIVATE auto_launch_win.cpp meta_output_win.cpp)
  trace_settings(dinput_vr_proxy)
  target_include_directories(dinput_vr_proxy PRIVATE "${ED_ROOT}/include")
  target_include_directories(dinput_vr_proxy SYSTEM PRIVATE "${ED_ROOT}/third_party/gameinput/4.0.306/native/include")
  target_compile_definitions(dinput_vr_proxy PRIVATE ED_VR_HAPTICS_MIRROR=1)
  target_link_libraries(dinput_vr_proxy PRIVATE bcrypt minhook authored_motion_output gameinput_compat ws2_32 sco_feedback)
  target_link_options(dinput_vr_proxy PRIVATE -Wl,--exclude-all-symbols)
  set_target_properties(dinput_vr_proxy PROPERTIES PREFIX "" OUTPUT_NAME dinput8 RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/experimental-proxy" ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/experimental-proxy")
  get_target_property(CHAIN_SOURCES dinput_restoration_chain_proxy SOURCES)
  list(TRANSFORM CHAIN_SOURCES PREPEND "${ED_ROOT}/")
  list(REMOVE_ITEM CHAIN_SOURCES "${ED_ROOT}/src/dinput8_chain.def")
  add_library(dinput_vr_chain_proxy SHARED ${CHAIN_SOURCES} publisher_win.cpp bridge_exports.cpp dinput_vr_chain.def)
  target_sources(dinput_vr_chain_proxy PRIVATE auto_launch_win.cpp meta_output_win.cpp)
  trace_settings(dinput_vr_chain_proxy)
  target_include_directories(dinput_vr_chain_proxy PRIVATE "${ED_ROOT}/include")
  target_include_directories(dinput_vr_chain_proxy SYSTEM PRIVATE "${ED_ROOT}/third_party/gameinput/4.0.306/native/include")
  target_compile_definitions(dinput_vr_chain_proxy PRIVATE ED_VR_HAPTICS_MIRROR=1 ED_DINPUT_EARLY_CHAIN=1)
  add_dependencies(dinput_vr_chain_proxy previous_dinput_import)
  target_link_libraries(dinput_vr_chain_proxy PRIVATE bcrypt minhook authored_motion_output gameinput_compat "${CMAKE_BINARY_DIR}/libed_previous_dinput.a" ws2_32 sco_feedback)
  target_link_options(dinput_vr_chain_proxy PRIVATE -Wl,--exclude-all-symbols)
  set_target_properties(dinput_vr_chain_proxy PROPERTIES PREFIX "" OUTPUT_NAME dinput8 RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/experimental-chain-proxy" ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/experimental-chain-proxy")
  if(BUILD_TESTING)
  add_executable(vr_mirror_verification "${ED_ROOT}/verification/vr_mirror.cpp"
    "${ED_ROOT}/src/xinput_output.cpp" "${ED_ROOT}/src/gameinput_output.cpp"
    "${ED_ROOT}/src/motor_ownership.cpp" "${ED_ROOT}/src/onfoot_haptics.cpp" "${ED_ROOT}/src/onfoot_runtime.cpp"
    "${ED_ROOT}/src/ship_weapon_haptics.cpp")
  trace_settings(vr_mirror_verification)
  target_include_directories(vr_mirror_verification PRIVATE "${ED_ROOT}/include")
  target_include_directories(vr_mirror_verification SYSTEM PRIVATE "${ED_ROOT}/third_party/gameinput/4.0.306/native/include")
  target_compile_definitions(vr_mirror_verification PRIVATE ED_VR_HAPTICS_MIRROR=1)
  target_link_libraries(vr_mirror_verification PRIVATE minhook authored_motion_output gameinput_compat)
  target_link_options(vr_mirror_verification PRIVATE -municode)
  add_executable(vr_publisher_verification "${ED_ROOT}/verification/vr_publisher.cpp" publisher_win.cpp)
  trace_settings(vr_publisher_verification)
  target_include_directories(vr_publisher_verification PRIVATE "${ED_ROOT}/include")
  target_link_libraries(vr_publisher_verification PRIVATE ws2_32)
  add_executable(vr_broker_verification "${ED_ROOT}/verification/vr_broker.cpp" publisher_win.cpp)
  trace_settings(vr_broker_verification)
  target_include_directories(vr_broker_verification PRIVATE "${ED_ROOT}/include")
  target_link_libraries(vr_broker_verification PRIVATE ws2_32)
  endif()
else()
  add_library(ed_vr_openxr_native SHARED openxr_layer.cpp native_xr_relay.cpp)
  target_include_directories(ed_vr_openxr_native PRIVATE "${ED_ROOT}/include")
  target_include_directories(ed_vr_openxr_native SYSTEM PRIVATE "${OPENXR_ROOT}/include")
  target_compile_options(ed_vr_openxr_native PRIVATE -Wall -Wextra -Wpedantic -Werror -fvisibility=hidden -pthread)
  target_link_libraries(ed_vr_openxr_native PRIVATE dl)
  target_link_options(ed_vr_openxr_native PRIVATE -pthread "-Wl,--version-script=${CMAKE_CURRENT_SOURCE_DIR}/openxr_layer.map")
  set_target_properties(ed_vr_openxr_native PROPERTIES OUTPUT_NAME ed-vr-openxr)
  target_link_directories(ed_vr_haptics PRIVATE "${CMAKE_BINARY_DIR}")
  target_link_libraries(ed_vr_haptics PRIVATE openvr_api)
  target_link_options(ed_vr_haptics PRIVATE "-Wl,-rpath,$ORIGIN")
  set_target_properties(ed_vr_haptics PROPERTIES SKIP_BUILD_RPATH TRUE)
  configure_file("${OPENVR_ROOT}/bin/linux64/libopenvr_api.so" libopenvr_api.so COPYONLY)
endif()
